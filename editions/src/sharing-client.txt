const captureKey = "editions.pending-link.v1";
const capture = document.getElementById("capture-link");
if (capture) {
  let title = "";
  let incomingURL = "";
  if (location.hash) {
    const fragment = location.hash.slice(1);
    const values = new URLSearchParams(fragment);
    incomingURL = fragment.startsWith("url=") ? values.get("url") : fragment;
    title = fragment.startsWith("url=") ? (values.get("title") || "").slice(0,240) : "";
    capture.elements.url.value = incomingURL;
    history.replaceState(null, "", location.pathname);
  }
  capture.addEventListener("submit", event => {
    event.preventDefault();
    try {
      const url = new URL(capture.elements.url.value);
      if (url.protocol !== "https:" || url.username || url.password ||
          /\/(private|private-feed|magic-link|unsubscribe|manage-subscription)(\/|$)/i.test(url.pathname) ||
          [...url.searchParams.keys()].some(key => /token|secret|password|signature|^key$|^sig$|^auth$|^email$|subscription_id/i.test(key)))
        throw new Error("Choose the public article permalink.");
      url.hash = "";
      sessionStorage.setItem(captureKey, JSON.stringify({url:url.href, title: capture.elements.url.value === incomingURL ? title : "", time:Date.now()}));
      location.assign("/studio?capture=1");
    } catch (error) {
      capture.querySelector("[data-capture-status]").textContent = error.name === "SecurityError" ? "Browser storage is unavailable. Copy the link and paste it in Studio." : "Choose a public HTTPS article permalink.";
    }
  });
}

const share = document.querySelector("[data-share-edition]");
if (share) share.addEventListener("click", async () => {
  const url = location.origin + location.pathname;
  const status = document.querySelector("[data-share-status]");
  try {
    if (navigator.share) await navigator.share({title:document.title, url});
    else { await navigator.clipboard.writeText(url); status.textContent = "Edition link copied."; }
  } catch (error) { if (error.name !== "AbortError") status.textContent = url; }
});
